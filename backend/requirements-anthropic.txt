#     AI_PROVIDER=anthropic

#     ANTHROPIC_MODEL=<one of the IDs check_anthropic.py printed>
#

# a version nobody has run. Pin it with the command above once it is installed.
#
# Note this is an ADDITION, not a replacement. `openai` stays in requirements.txt
# because OpenAIProvider is still there and still selectable with AI_PROVIDER=openai;
